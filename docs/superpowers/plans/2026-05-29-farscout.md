# farscout Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a free, mostly-autonomous Farcaster research agent that reads Farcaster via HAATZ, reasons with OpenRouter free models (routing heavy/private work to a local Mac Ollama when reachable), stores learnings in the ZABAL Bonfire knowledge graph, and converses with the operator in Discord on an engagement-scaled cadence.

**Architecture:** Single long-running Node.js process on bot-hosting.net (256MB RAM). Seven focused modules: orchestrator (`index.js`) plus `lib/{cadence,config,reader,brain,memory,research,discord}.js`. Pure-logic modules (cadence, config parsing) are fully unit-tested; I/O modules are tested against mocked fetch.

**Tech Stack:** Node.js 20+ (ESM), `discord.js`, `undici` (fetch), `dotenv`. Test runner: Node built-in `node --test` (zero extra deps). State persisted as JSON files in `state/`.

```js
export function makeReader({ base, fid, fetchImpl, neynarKey = '' }) {
  async function get(path) {
    const url = `${base}${path}`;
    try {
      const res = await fetchImpl(url, { headers: { accept: 'application/json' } });
      if (!res.ok) {
        if (neynarKey) {
          const nres = await fetchImpl(`https://api.neynar.com${path}`, { headers: { api_key: neynarKey, accept: 'application/json' } });
          if (nres.ok) return nres.json();
        }
        return null;
      }
      return res.json();
    } catch {
      return null;
    }
  }
  return {
    async userCasts(limit = 25) {
      const d = await get(`/v2/farcaster/feed/user/casts?fid=${fid}&limit=${limit}`);
      return d?.casts ?? [];
    },
    async channelFeed(channels, limit = 25) {
      const d = await get(`/v2/farcaster/feed/channels?channel_ids=${channels.join(',')}&limit=${limit}`);
      return d?.casts ?? [];
    },
    async bulkUsers(fids) {
      const d = await get(`/v2/farcaster/user/bulk?fids=${fids.join(',')}`);
      return d?.users ?? [];
    },
  };
}
```

```js
export function makeBrain({ openrouterKey, freeModels, ollamaUrl, ollamaModel = 'llama3.1', fetchImpl }) {
  async function ollamaReachable() {
    if (!ollamaUrl) return false;
    try {
      const res = await fetchImpl(`${ollamaUrl}/api/tags`, { method: 'GET' });
      return res.ok;
    } catch { return false; }
  }
  async function askOllama(prompt) {
    const res = await fetchImpl(`${ollamaUrl}/api/chat`, {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ model: ollamaModel, stream: false, messages: [{ role: 'user', content: prompt }] }),
    });
    if (!res.ok) return null;
    const d = await res.json();
    return d?.message?.content ?? null;
  }
  async function askCloud(prompt) {
    for (const model of freeModels) {
      try {
        const res = await fetchImpl('https://openrouter.ai/api/v1/chat/completions', {
          method: 'POST',
          headers: { Authorization: `Bearer ${openrouterKey}`, 'content-type': 'application/json' },
          body: JSON.stringify({ model, messages: [{ role: 'user', content: prompt }] }),
        });
        if (!res.ok) continue;
        const d = await res.json();
        const text = d?.choices?.[0]?.message?.content;
        if (text) return text;
      } catch { /* next model */ }
    }
    return null;
  }
  return {
    async ask(prompt, { tier = 'light' } = {}) {
      if ((tier === 'heavy' || tier === 'private') && (await ollamaReachable())) {
        const local = await askOllama(prompt);
        if (local) return local;
      }
      const cloud = await askCloud(prompt);
      if (cloud) return cloud;
      if (await ollamaReachable()) return askOllama(prompt);
      return null;
    },
  };
}
```

```js
import { readFile, writeFile } from 'node:fs/promises';

export function makeMemory({ file, bonfireKey = '', bonfireBase = '', fetchImpl }) {
  let state = { episodeHashes: [], queue: [] };
  const known = new Set();
  async function persist() { await writeFile(file, JSON.stringify(state, null, 2)); }
  return {
    async load() {
      try {
        state = JSON.parse(await readFile(file, 'utf8'));
      } catch { state = { episodeHashes: [], queue: [] }; }
      for (const h of state.episodeHashes) known.add(h);
    },
    isKnown(key) { return known.has(key); },
    remember(key) { if (!known.has(key)) { known.add(key); state.episodeHashes.push(key); } },
    queueSize() { return state.queue.length; },
    async pushEpisode(text, meta = {}) {
      const body = JSON.stringify({ content: text, metadata: meta });
      try {
        const res = await fetchImpl(`${bonfireBase}/api/episodes`, {
          method: 'POST',
          headers: { 'content-type': 'application/json', authorization: `Bearer ${bonfireKey}` },
          body,
        });
        if (res.ok) { await persist(); return true; }
      } catch { /* fall through */ }
      state.queue.push({ text, meta });
      await persist();
      return false;
    },
    async flushQueue() {
      const pending = state.queue.splice(0);
      for (const item of pending) {
        const ok = await this.pushEpisode(item.text, item.meta);
        if (!ok) break;
      }
      await persist();
    },
    async save() { await persist(); },
  };
}
```

```js
function parseJson(text, fallback) {
  if (!text) return fallback;
  const m = text.match(/\{[\s\S]*\}/);
  if (!m) return fallback;
  try { return JSON.parse(m[0]); } catch { return fallback; }
}

export async function runCycle({ reader, brain, memory, channels = [], recentReplies = [] }) {
  const casts = await reader.userCasts(25);
  const channelCasts = channels.length ? await reader.channelFeed(channels, 25) : [];
  const corpus = [...casts, ...channelCasts].map((c) => c.text).filter(Boolean).join('\n');
  if (!corpus) return { findings: [], questions: [] };

  const extract = await brain.ask(
    `Summarize what this Farcaster operator and their channels are focused on. Reply ONLY JSON: {"topics":["kebab-topic"],"summary":"..."}.\n\n${corpus}`,
    { tier: 'light' },
  );
  const { topics = [] } = parseJson(extract, { topics: [] });
  const novel = topics.filter((t) => !memory.isKnown(t));
  if (!novel.length) return { findings: [], questions: [] };

  const replyContext = recentReplies.length ? `\nOperator recently said: ${recentReplies.join(' | ')}` : '';
  const research = await brain.ask(
    `You are a Farcaster research scout. Research these novel topics and report. Reply ONLY JSON: {"findings":["..."],"questions":["max 2 short questions for the operator"]}.\nTopics: ${novel.join(', ')}${replyContext}`,
    { tier: 'heavy' },
  );
  const { findings = [], questions = [] } = parseJson(research, { findings: [], questions: [] });

  for (const t of novel) memory.remember(t);
  for (const f of findings) await memory.pushEpisode(f, { source: 'farscout', topics: novel });

  return { findings, questions: questions.slice(0, 2) };
}
```

```js
import { Client, GatewayIntentBits, Partials } from 'discord.js';

export function makeDiscord({ token, userId, onCommand }) {
  const client = new Client({
    intents: [GatewayIntentBits.DirectMessages, GatewayIntentBits.Guilds, GatewayIntentBits.MessageContent, GatewayIntentBits.GuildMessages],
    partials: [Partials.Channel],
  });
  let engaged = false;
  const replies = [];

  client.on('messageCreate', (msg) => {
    if (msg.author.bot) return;
    if (msg.author.id !== userId) return;
    const text = msg.content.trim();
    if (text.startsWith('/')) {
      const [cmd, ...rest] = text.slice(1).split(' ');
      onCommand?.(cmd, rest.join(' '), msg);
      return;
    }
    engaged = true;
    replies.push(text);
    if (replies.length > 10) replies.shift();
  });

  return {
    async start() { await client.login(token); await new Promise((r) => client.once('clientReady', r)); },
    async deliver(text) {
      const user = await client.users.fetch(userId);
      const chunks = text.match(/[\s\S]{1,1900}/g) ?? [];
      for (const c of chunks) await user.send(c);
    },
    consumeEngagement() { const e = engaged; engaged = false; return e; },
    recentReplies() { return [...replies]; },
    clearReplies() { replies.length = 0; },
  };
}
```

```js
    userId: config.discordUserId,
    onCommand: async (cmd) => {
      if (cmd === 'pause') { paused = true; await discord.deliver('Paused.'); }
      if (cmd === 'resume') { paused = false; await discord.deliver('Resumed.'); }
      if (cmd === 'now') { await tick(); }
    },
  });

  await discord.deliver('farscout online. Watching Farcaster. Reply any time to speed me up.');

  let { interval } = await loadCadence(CADENCE_FILE, START_MS);

  async function tick() {
    if (paused) return;
    await memory.flushQueue();
    const out = await runCycle({ reader, brain, memory, channels: config.watchChannels, recentReplies: discord.recentReplies() });
    if (out.findings.length || out.questions.length) {
      const msg = [out.findings.length ? `Found:\n- ${out.findings.join('\n- ')}` : '', out.questions.length ? `\nQuestions:\n- ${out.questions.join('\n- ')}` : ''].join('\n');
      await discord.deliver(msg.trim());
    }
    const engaged = discord.consumeEngagement();
    interval = nextInterval(interval, engaged);
    await saveCadence(CADENCE_FILE, { interval });
    schedule();
  }

  let timer;
  function schedule() { clearTimeout(timer); timer = setTimeout(tick, interval); }
  schedule();
}
```

```js
import { fetch } from 'undici';
import { join, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';
import { config, requireConfig } from '../config.js';
import { makeBrain } from '../lib/brain.js';
import { makeMemory } from '../lib/memory.js';

const ROOT = dirname(dirname(fileURLToPath(import.meta.url)));

  'Farcaster Mini Apps integration (doc 173)',
  'Farcaster Mini Apps llms.txt 2026 (doc 250)',

  'Farcaster Snaps (doc 295)',

  requireConfig();
  const brain = makeBrain({ openrouterKey: config.openrouterKey, freeModels: config.freeModels, ollamaUrl: config.ollamaUrl, fetchImpl: fetch });
  const memory = makeMemory({ file: join(ROOT, 'state', 'cache.json'), bonfireKey: config.bonfireKey, bonfireBase: config.bonfireBase, fetchImpl: fetch });
```
